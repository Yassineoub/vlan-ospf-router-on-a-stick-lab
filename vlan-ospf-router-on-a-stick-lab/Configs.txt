VLAN plan:  IT = 10 | HR = 20 | Sales = 30 | Accounting = 40


===========================================================
 1. R1
===========================================================
enable
configure terminal
hostname R1


! --- Link to SW0 (trunk side): sub-interfaces, one per VLAN ---
interface GigabitEthernet0/0
 no shutdown
 exit

interface GigabitEthernet0/0.10
 description IT gateway
 encapsulation dot1Q 10
 ip address 192.168.1.1 255.255.255.192
 exit

interface GigabitEthernet0/0.20
 description HR gateway
 encapsulation dot1Q 20
 ip address 192.168.1.65 255.255.255.192
 exit

interface GigabitEthernet0/0.30
 description Sales gateway
 encapsulation dot1Q 30
 ip address 192.168.1.129 255.255.255.224
 exit

interface GigabitEthernet0/0.40
 description Accounting gateway
 encapsulation dot1Q 40
 ip address 192.168.1.161 255.255.255.240
 exit

! --- Link to SW5 / Server0 ---
interface GigabitEthernet0/1
 description Link to server network
 ip address 172.16.1.1 255.255.255.252
 no shutdown
 exit

! --- OSPF ---
router ospf 1
 router-id 1.1.1.1
 network 192.168.1.0 0.0.0.63 area 0
 network 192.168.1.64 0.0.0.63 area 0
 network 192.168.1.128 0.0.0.31 area 0
 network 192.168.1.160 0.0.0.15 area 0
 network 172.16.1.0 0.0.0.3 area 0
 exit

end
write memory


===========================================================
 2. SW0 (core switch)
===========================================================
! Fa0/1 -> R1 | Fa0/2 -> SW1 | Fa0/3 -> SW2 | Fa0/4 -> SW3 | Fa0/5 -> SW4
enable
configure terminal
hostname SW0


vlan 10
 name IT
vlan 20
 name HR
vlan 30
 name Sales
vlan 40
 name Accounting
 exit

interface range FastEthernet0/1 - 5
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 no shutdown
 exit

end
write memory

===========================================================
 End-device addressing (static)
===========================================================
Each PC and the server uses a static IP from its subnet, the subnet
mask of its subnet, and the department gateway as default gateway.

Dept        Mask               Default gateway
IT          255.255.255.192    192.168.1.1
HR          255.255.255.192    192.168.1.65
Sales       255.255.255.224    192.168.1.129
Accounting  255.255.255.240    192.168.1.161
Server0     255.255.255.252    172.16.1.1   (Server0 IP: 172.16.1.2)
